# Example curl commands for testing all vehicle CRUD endpoints
# Assumes API is running at http://localhost:4000
docker compose down
docker rm -f codespaces-postgres
docker compose up -d
# Create a vehicle (POST)
curl -X POST http://localhost:4000/v1/vehicles \
  -H "Content-Type: application/json" \
  -d '{
    "vin": "1HGCM82633A004319",
    "make": "Honda",
    "model": "Civic",
    "year": 2020,
    "category": "Sedan",
    "daily_rate": 49,
    "mileage": 10000,
    "fuel_capacity": 50,
    "status": "available"
  }'

# List all vehicles (GET)
curl http://localhost:4000/v1/vehicles

# Get a vehicle by ID (GET)
curl http://localhost:4000/v1/vehicles/7

# Update a vehicle completely (PUT)
curl -X PUT http://localhost:4000/v1/vehicles/7 \
  -H "Content-Type: application/json" \
  -d '{
    "vin": "1KGCM82633A004352",
    "make": "Honda",
    "model": "Accord",
    "year": 2021,
    "category": "Sedan",
    "daily_rate": 59.99,
    "mileage": 12000,
    "fuel_capacity": 50,
    "status": "rented"
  }'

# Partially update a vehicle (PATCH)
curl -X PATCH http://localhost:4000/v1/vehicles/2 \
  -H "Content-Type: application/json" \
  -d '{
    "status": "maintenance",
    "mileage": 12500
  }'

# Delete a vehicle (DELETE)
curl -X DELETE http://localhost:4000/v1/vehicles/7
